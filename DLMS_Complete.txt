१. Base & Helpers (मूलभूत सहाय्यक फाईल्स) - Helpers फोल्डरमध्ये:

Imports System.Data.SQLite
Imports System.IO

Public Class DatabaseHelper
    Public Shared ReadOnly ConnectionString As String = "Data Source=" & Path.Combine(Application.StartupPath, "DLMS.db") & ";Version=3;"

    Public Shared Sub InitializeDatabase()
        If Not File.Exists(Path.Combine(Application.StartupPath, "DLMS.db")) Then
            Using conn As New SQLiteConnection(ConnectionString)
                conn.Open()
                Using cmd As New SQLiteCommand("PRAGMA journal_mode=WAL; PRAGMA foreign_keys=ON;", conn)
                    cmd.ExecuteNonQuery()
                End Using

                Dim schema As String = "
                    CREATE TABLE tblUsers (UserId INTEGER PRIMARY KEY AUTOINCREMENT, Username TEXT UNIQUE NOT NULL, PasswordHash TEXT NOT NULL, Role TEXT NOT NULL, IsActive INTEGER DEFAULT 1);
                    CREATE TABLE tblDoctors (DoctorId INTEGER PRIMARY KEY AUTOINCREMENT, DoctorName TEXT NOT NULL, ClinicName TEXT, Mobile TEXT, Address TEXT, City TEXT, Email TEXT, GST TEXT, OpeningBalance DECIMAL(10,2) DEFAULT 0, CreditLimit DECIMAL(10,2) DEFAULT 0, Status TEXT DEFAULT 'Active');
                    CREATE TABLE tblPatients (PatientId INTEGER PRIMARY KEY AUTOINCREMENT, PatientName TEXT NOT NULL, Age INTEGER, Gender TEXT, DoctorId INTEGER, Mobile TEXT, History TEXT, FOREIGN KEY(DoctorId) REFERENCES tblDoctors(DoctorId));
                    CREATE TABLE tblWorkTypes (WorkId INTEGER PRIMARY KEY AUTOINCREMENT, WorkName TEXT UNIQUE NOT NULL, Category TEXT, Price DECIMAL(10,2) NOT NULL, Status TEXT DEFAULT 'Active');
                    CREATE TABLE tblEntries (EntryId INTEGER PRIMARY KEY AUTOINCREMENT, EntryNumber TEXT UNIQUE NOT NULL, EntryDate TEXT NOT NULL, DoctorId INTEGER, PatientId INTEGER, ToothNumber TEXT, WorkId INTEGER, Models TEXT, Remarks TEXT, DeliveryDate TEXT, Amount DECIMAL(10,2) NOT NULL, Status INTEGER DEFAULT 0, IsDeleted INTEGER DEFAULT 0, FOREIGN KEY(DoctorId) REFERENCES tblDoctors(DoctorId), FOREIGN KEY(PatientId) REFERENCES tblPatients(PatientId), FOREIGN KEY(WorkId) REFERENCES tblWorkTypes(WorkId));
                    CREATE TABLE tblBills (BillId INTEGER PRIMARY KEY AUTOINCREMENT, BillNumber TEXT UNIQUE NOT NULL, BillDate TEXT NOT NULL, DoctorId INTEGER, PatientId INTEGER, SubTotal DECIMAL(10,2) NOT NULL, Discount DECIMAL(10,2) DEFAULT 0, GrandTotal DECIMAL(10,2) NOT NULL, Balance DECIMAL(10,2) NOT NULL, Status TEXT DEFAULT 'Unpaid', FOREIGN KEY(DoctorId) REFERENCES tblDoctors(DoctorId), FOREIGN KEY(PatientId) REFERENCES tblPatients(PatientId));
                    CREATE TABLE tblBillDetails (DetailId INTEGER PRIMARY KEY AUTOINCREMENT, BillId INTEGER NOT NULL, EntryId INTEGER NOT NULL, Amount DECIMAL(10,2) NOT NULL, FOREIGN KEY(BillId) REFERENCES tblBills(BillId), FOREIGN KEY(EntryId) REFERENCES tblEntries(EntryId));
                    CREATE TABLE tblPayments (PaymentId INTEGER PRIMARY KEY AUTOINCREMENT, BillId INTEGER NOT NULL, PaymentDate TEXT NOT NULL, Mode TEXT NOT NULL, ReferenceNo TEXT, Amount DECIMAL(10,2) NOT NULL, Remarks TEXT, FOREIGN KEY(BillId) REFERENCES tblBills(BillId));
                    CREATE TABLE tblBackupHistory (BackupId INTEGER PRIMARY KEY AUTOINCREMENT, BackupDate TEXT NOT NULL, FilePath TEXT NOT NULL, Size TEXT, Type TEXT);
                    CREATE INDEX idx_entries_doctor ON tblEntries(DoctorId);
                    CREATE INDEX idx_entries_status ON tblEntries(Status);
                    CREATE INDEX idx_payments_bill ON tblPayments(BillId);
                    CREATE INDEX idx_bills_doctor ON tblBills(DoctorId);
                "
                Using cmd As New SQLiteCommand(schema, conn)
                    cmd.ExecuteNonQuery()
                End Using
                Using cmd As New SQLiteCommand("INSERT INTO tblUsers (Username, PasswordHash, Role, IsActive) VALUES (@u, @p, 'Admin', 1)", conn)
                    cmd.Parameters.AddWithValue("@u", "admin")
                    cmd.Parameters.AddWithValue("@p", BCrypt.Net.BCrypt.HashPassword("admin123"))
                    cmd.ExecuteNonQuery()
                End Using
            End Using
        End If
    End Sub
End Class

' File: SQLiteDataAccess.vb (DataAccess)
' ================================================================
Imports System.Data.SQLite
Imports System.IO

Public Class SQLiteDataAccess
    Implements IDisposable
    Private ReadOnly _connection As SQLiteConnection
    Public Sub New()
        _connection = New SQLiteConnection(DatabaseHelper.ConnectionString)
    End Sub
    Public Function OpenConnection() As SQLiteConnection
        If _connection.State <> ConnectionState.Open Then _connection.Open()
        Return _connection
    End Function
    Public Function ExecuteQuery(sql As String, Optional parameters As Dictionary(Of String, Object) = Nothing) As DataTable
        Dim dt As New DataTable()
        Using cmd As New SQLiteCommand(sql, OpenConnection())
            If parameters IsNot Nothing Then
                For Each kvp In parameters
                    cmd.Parameters.AddWithValue(kvp.Key, kvp.Value)
                Next
            End If
            Using reader As SQLiteDataReader = cmd.ExecuteReader()
                dt.Load(reader)
            End Using
        End Using
        Return dt
    End Function
    Public Function ExecuteNonQuery(sql As String, Optional parameters As Dictionary(Of String, Object) = Nothing) As Integer
        Using cmd As New SQLiteCommand(sql, OpenConnection())
            If parameters IsNot Nothing Then
                For Each kvp In parameters
                    cmd.Parameters.AddWithValue(kvp.Key, kvp.Value)
                Next
            End If
            Return cmd.ExecuteNonQuery()
        End Using
    End Function
    Public Function ExecuteScalar(Of T)(sql As String, Optional parameters As Dictionary(Of String, Object) = Nothing) As T
        Using cmd As New SQLiteCommand(sql, OpenConnection())
            If parameters IsNot Nothing Then
                For Each kvp In parameters
                    cmd.Parameters.AddWithValue(kvp.Key, kvp.Value)
                Next
            End If
            Dim result = cmd.ExecuteScalar()
            If result IsNot Nothing AndAlso result IsNot DBNull.Value Then Return CType(result, T)
            Return Nothing
        End Using
    End Function
    Public Sub CloseConnection()
        If _connection.State = ConnectionState.Open Then _connection.Close()
    End Sub
    Public Sub Dispose() Implements IDisposable.Dispose
        CloseConnection()
        _connection.Dispose()
    End Sub
End Class

Public Class DataSeeder
    Public Shared Sub SeedInitialData()
        Using dal As New SQLiteDataAccess()
            Dim count As Integer = dal.ExecuteScalar(Of Integer)("SELECT COUNT(*) FROM tblWorkTypes")
            If count = 0 Then
                Dim workTypes As String() = {"Zirconia", "PFM", "Metal", "Wax", "Implant", "CAD CAM", "Laser", "Denture", "Temporary Crown", "3D Print"}
                Dim prices As Decimal() = {1500, 1200, 800, 500, 2500, 3000, 2000, 1800, 600, 2200}
                For i As Integer = 0 To workTypes.Length - 1
                    Dim sql As String = "INSERT INTO tblWorkTypes (WorkName, Category, Price, Status) VALUES (@Name, 'General', @Price, 'Active')"
                    Dim params As New Dictionary(Of String, Object) From {{"@Name", workTypes(i)}, {"@Price", prices(i)}}
                    dal.ExecuteNonQuery(sql, params)
                Next
            End If
            Dim docCount As Integer = dal.ExecuteScalar(Of Integer)("SELECT COUNT(*) FROM tblDoctors")
            If docCount = 0 Then
                Dim sql As String = "INSERT INTO tblDoctors (DoctorName, ClinicName, Mobile, City, Status) VALUES (@Name, 'Sunrise Dental Clinic', '9876543210', 'Mumbai', 'Active')"
                Dim params As New Dictionary(Of String, Object) From {{"@Name", "Dr. Sharma"}}
                dal.ExecuteNonQuery(sql, params)
            End If
        End Using
    End Sub
End Class

Imports System.Data.SQLite
Imports System.IO

Public Class BackupManager
    Public Shared Sub PerformBackup()
        Try
            Dim sourceDb As String = Path.Combine(Application.StartupPath, "DLMS.db")
            Dim backupFolder As String = Path.Combine(Application.StartupPath, "Backups")
            If Not File.Exists(sourceDb) Then Return
            If Not Directory.Exists(backupFolder) Then Directory.CreateDirectory(backupFolder)
            Dim timestamp As String = DateTime.Now.ToString("yyyyMMdd_HHmmss")
            Dim destFile As String = Path.Combine(backupFolder, $"DLMS_Backup_{timestamp}.db")
            Using conn As New SQLiteConnection(DatabaseHelper.ConnectionString)
                conn.Open()
                Using cmd As New SQLiteCommand($"VACUUM INTO '{destFile}';", conn)
                    cmd.ExecuteNonQuery()
                End Using
            End Using
            Using dal As New SQLiteDataAccess()
                Dim sql As String = "INSERT INTO tblBackupHistory (BackupDate, FilePath, Size, Type) VALUES (@Date, @Path, @Size, 'Auto')"
                Dim fi As New FileInfo(destFile)
                Dim params As New Dictionary(Of String, Object) From {{"@Date", DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss")}, {"@Path", destFile}, {"@Size", fi.Length.ToString()}}
                dal.ExecuteNonQuery(sql, params)
            End Using
        Catch ex As Exception
            File.WriteAllText(Path.Combine(Application.StartupPath, "Backup_Error.log"), $"{DateTime.Now}: {ex.Message}")
        End Try
    End Sub
End Class

' File: DoctorRepository.vb (DataAccess)

Public Class DoctorRepository
    Implements IDisposable
    Private ReadOnly _dal As SQLiteDataAccess
    Public Sub New()
        _dal = New SQLiteDataAccess()
    End Sub
    Public Function GetAllDoctors() As List(Of Doctor)
        Dim dt As DataTable = _dal.ExecuteQuery("SELECT * FROM tblDoctors WHERE Status = 'Active' ORDER BY DoctorName")
        Dim list As New List(Of Doctor)()
        For Each row As DataRow In dt.Rows
            list.Add(New Doctor() With {
                .DoctorId = CInt(row("DoctorId")),
                .DoctorName = row("DoctorName").ToString(),
                .ClinicName = row("ClinicName").ToString(),
                .Mobile = row("Mobile").ToString(),
                .City = row("City").ToString(),
                .OpeningBalance = CDec(row("OpeningBalance")),
                .Status = row("Status").ToString()
            })
        Next
        Return list
    End Function
    Public Function AddDoctor(doc As Doctor) As Boolean
        Dim sql As String = "INSERT INTO tblDoctors (DoctorName, ClinicName, Mobile, Address, City, Email, GST, OpeningBalance, CreditLimit, Status) VALUES (@Name, @Clinic, @Mobile, @Address, @City, @Email, @GST, @OB, @CL, @Status)"
        Dim params As New Dictionary(Of String, Object) From {
            {"@Name", doc.DoctorName}, {"@Clinic", doc.ClinicName}, {"@Mobile", doc.Mobile}, {"@Address", doc.Address},
            {"@City", doc.City}, {"@Email", doc.Email}, {"@GST", doc.GST}, {"@OB", doc.OpeningBalance}, {"@CL", doc.CreditLimit},
            {"@Status", doc.Status}
        }
        Return _dal.ExecuteNonQuery(sql, params) > 0
    End Function
    Public Sub Dispose() Implements IDisposable.Dispose
        _dal.Dispose()
    End Sub
End Class

' File: PatientRepository.vb (DataAccess)
' ================================================================
Public Class PatientRepository
    Implements IDisposable
    Private ReadOnly _dal As SQLiteDataAccess
    Public Sub New()
        _dal = New SQLiteDataAccess()
    End Sub
    Public Function GetAllPatients() As List(Of Patient)
        Dim dt As DataTable = _dal.ExecuteQuery("SELECT * FROM tblPatients ORDER BY PatientName")
        Return MapDataTableToPatients(dt)
    End Function
    Public Function GetPatientsByDoctor(doctorId As Integer) As List(Of Patient)
        Dim sql As String = "SELECT * FROM tblPatients WHERE DoctorId = @DocId ORDER BY PatientName"
        Dim params As New Dictionary(Of String, Object) From {{"@DocId", doctorId}}
        Dim dt As DataTable = _dal.ExecuteQuery(sql, params)
        Return MapDataTableToPatients(dt)
    End Function
    Public Function AddPatient(patient As Patient) As Boolean
        Dim checkSql As String = "SELECT COUNT(*) FROM tblPatients WHERE Mobile = @Mobile"
        Dim checkParams As New Dictionary(Of String, Object) From {{"@Mobile", patient.Mobile}}
        If _dal.ExecuteScalar(Of Integer)(checkSql, checkParams) > 0 Then
            Throw New Exception("Duplicate Mobile Number found!")
        End If
        Dim sql As String = "INSERT INTO tblPatients (PatientName, Age, Gender, DoctorId, Mobile, History) VALUES (@Name, @Age, @Gender, @DocId, @Mobile, @History)"
        Dim params As New Dictionary(Of String, Object) From {
            {"@Name", patient.PatientName}, {"@Age", If(patient.Age.HasValue, patient.Age.Value, DBNull.Value)},
            {"@Gender", patient.Gender}, {"@DocId", patient.DoctorId}, {"@Mobile", patient.Mobile}, {"@History", patient.History}
        }
        Return _dal.ExecuteNonQuery(sql, params) > 0
    End Function
    Public Sub Dispose() Implements IDisposable.Dispose
        _dal.Dispose()
    End Sub
    Private Function MapDataTableToPatients(dt As DataTable) As List(Of Patient)
        Dim list As New List(Of Patient)()
        For Each row As DataRow In dt.Rows
            list.Add(New Patient() With {
                .PatientId = CInt(row("PatientId")), .PatientName = row("PatientName").ToString(),
                .Age = If(row("Age") Is DBNull.Value, Nothing, CInt(row("Age"))), .Gender = row("Gender").ToString(),
                .DoctorId = CInt(row("DoctorId")), .Mobile = row("Mobile").ToString(), .History = row("History").ToString()
            })
        Next
        Return list
    End Function
End Class

Public Class DoctorService
    Implements IDisposable
    Private ReadOnly _repo As DoctorRepository
    Public Sub New()
        _repo = New DoctorRepository()
    End Sub
    Public Function RegisterNewDoctor(doc As Doctor) As Boolean
        If String.IsNullOrWhiteSpace(doc.DoctorName) Then Throw New ArgumentException("Doctor Name is required.")
        If String.IsNullOrWhiteSpace(doc.Mobile) Then Throw New ArgumentException("Mobile Number is required.")
        If Not doc.Mobile.All(AddressOf Char.IsDigit) Then Throw New ArgumentException("Mobile must contain only digits.")
        Return _repo.AddDoctor(doc)
    End Function
    Public Function GetAllActiveDoctors() As List(Of Doctor)
        Return _repo.GetAllDoctors()
    End Function
    Public Sub Dispose() Implements IDisposable.Dispose
        _repo.Dispose()
    End Sub
End Class
